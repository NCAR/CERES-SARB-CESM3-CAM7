\section{AOD Constraints}
\begin{frame}[t]
\frametitle{AOD Constraints}
\footnotesize
\begin{block}{Uniform Column Rescale at 550 nm}
\vspace{-0.6em}
\[
  \tau_{\mathrm{m}} = \sum_{k}\sum_{m}
      \sigma_{\mathrm{ext},m}\!\left(r_{\mathrm{wet}}, n_{\mathrm{wet}}\right) M_{m,k},
  \qquad
  r = \operatorname{clamp}\!\left(\tau_{\mathrm{obs}}/\tau_{\mathrm{m}},\, 0.1,\, 10\right)
\]
\vspace{-1.4em}
\[
  q_{s,k} \leftarrow r\, q_{s,k}, \qquad N_{m,k} \leftarrow r\, N_{m,k}
\]
{\scriptsize\raggedright
$M_{m,k}$ aerosol mass per area of mode $m$ in layer $k$;
$\sigma_{\mathrm{ext},m}$ its mass extinction cross section at wet radius $r_{\mathrm{wet}}$ and refractive index $n_{\mathrm{wet}}$;
$\tau_{\mathrm{obs}}$ observed AOD;
$q_{s,k}$ species mass and $N_{m,k}$ mode number mixing ratios, all species, modes and levels.
$r = 1$ where $\tau_{\mathrm{m}} < 0.01$ or no observation.\par}
\vspace{-0.3em}
\end{block}

  \item Keeps speciation, size and shape;\\direct effect only
\end{itemize}
\end{block}

  \item First guess carries earlier increments
\end{itemize}
