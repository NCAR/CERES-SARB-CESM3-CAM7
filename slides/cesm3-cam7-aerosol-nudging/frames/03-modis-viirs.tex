\begin{block}{Products}
\begin{itemize}
  \item MODIS Terra and Aqua, Collection 6.1,\\Dark Target + Deep Blue, 550 nm
  \item VIIRS NOAA-20 Deep Blue\\(2018 on; 2008 is MODIS only)
  \item Level-2 swaths from the LAADS archive
\end{itemize}

\begin{block}{Screening and Gridding}
\begin{itemize}
  \item Quality-flagged retrievals only,\\$0 \le \tau \le 5$
  \item Pixels binned into f09 cells\\every 3 hours
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Merging}
\begin{itemize}
  \item Pixels averaged within each cell,\\sensors combined every 3 hours
\end{itemize}
\end{block}
\begin{block}{Gridded Product}
\begin{itemize}
  \item One global field per 3 h:\\AOD and pixel count
  \item No retrieval in a cell\\means the model column runs free
\end{itemize}
\end{block}
\end{column}
